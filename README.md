print("===== DailyMate =====")
print("1. ตารางงาน")
print("2. รายรับ-รายจ่าย")
print("3. บันทึกอารมณ์")
print("4. ออกจากระบบ")

menu = int(input("เลือกเมนู: "))

if menu == 1:
    print("คุณเลือก: ตารางงาน")
    task = input("กรอกชื่องาน: ")
    print("บันทึกงานเรียบร้อย:", task)

elif menu == 2:
    print("คุณเลือก: รายรับ-รายจ่าย")
    money = float(input("กรอกจำนวนเงิน: "))
    print("บันทึกข้อมูลเรียบร้อย:", money, "บาท")

elif menu == 3:
    print("คุณเลือก: บันทึกอารมณ์")
    mood = input("วันนี้คุณรู้สึกอย่างไร: ")
    print("บันทึกอารมณ์เรียบร้อย:", mood)

elif menu == 4:
    print("ขอบคุณที่ใช้ DailyMate")

else:
    print("กรุณาเลือกเมนู 1-4")
